\subsection{概述}
\begin{frame}{\insertsection~~\insertsubsection}
  \small
  \begin{itemize}
    \item 梵语有三种过去时，含义相同，可互换：\\
    \begin{itemize}
      \item 不完全过去时：增音符 + \important{现在时语干} + 次要语尾
      \item 完成时（第27--28章）：重复 + \important{完成时语干} + 完成时语尾
      \item \veryimportant{不定过去时}：增音符 + \important{不定过去时语干} + 次要语尾
    \end{itemize}
    \item 三者意义相同；不定过去时最少见。
    \item 识别口诀：\\
    增音符 + 次要语尾 + \veryimportant{非现在时语干} → 不定过去时
    \item 最重要用法：\skt{mā} + \veryimportant{不含增音符}的不定过去时\\
    = 禁止命令
  \end{itemize}
\end{frame}

\subsection{七种类型}
\begin{frame}{\insertsection~~七种类型}
  \small
  共同特征：增音符 \pratyaya{a-} + 次要语尾（同不完全过去时）\\
  区别在于\important{语干构成方式}：
  \begin{itemize}
    \item[1a)] \important{词根不定过去时}~~直接在词根上加语尾\\
    \notsoimportant{（限 \verbroot{bhū} 和 ā 结尾词根）}
    \item[1b)] \important{a-不定过去时}~~词根零级 + \pratyaya{-a-}~~\notsoimportant{（最常见）}
    \item[2)] \important{重复不定过去时}~~重复词根首辅音~~\notsoimportant{（派生动词）}
    \item[3a)] \important{s-不定过去时}~~词根 + \pratyaya{-s-}~~主动三合，中间零级
    \item[3b)] \important{iṣ-不定过去时}~~词根 + \pratyaya{-iṣ-}
    \item[3c)] \notsoimportant{siṣ-不定过去时~~词根 + -siṣ-~~（少见）}
    \item[3d)] \notsoimportant{sa-不定过去时~~词根零级 + -sa-~~（少见）}
  \end{itemize}
\end{frame}

\subsection{a-不定过去时}
\begin{frame}{\insertsection~~\insertsubsection~~（1b，最常见）}
  \small
  \begin{itemize}
    \item 构成：增音符 + \important{词根零级} + \pratyaya{-a-} + 次要语尾
    \item 形态上最接近不完全过去时，但语干不同。
    \item 词根不出现在字典中的形式：\\
    \verbroot{vid} (2) 'know' → 零级 \verbstem{vid-}\\
    ~~P.1st.Sg. \fullpada{avidam} ~~3rd.Pl. \fullpada{avidan}\\
    \verbroot{śuc} (1) 'grieve' → \fullpada{aśucat} \\
    \verbroot{kṛ} (8) 'do' → \fullpada{akarot} ~~3rd.Pl. \fullpada{akurvan}\\
    \item 注意：词根有零级变化时要辨认\\
    \verbroot{gam} → 零级 \verbstem{gam-}：\fullpada{agamat}\\
    \verbroot{vac} → 零级 \verbstem{uc-}：\fullpada{avocat}
  \end{itemize}
\end{frame}

\subsection{s-不定过去时}
\begin{frame}{\insertsection~~\insertsubsection~~（3a）}
  \small
  \begin{itemize}
    \item 构成：增音符 + 词根 + \pratyaya{-s-} + 次要语尾
    \item \important{主动语态}：词根\veryimportant{三合}（vṛddhi）\\
    \verbroot{rudh} → 三合 \verbstem{raudh-} + s → \fullpada{arautsam}
    \item \important{中间语态}：词根\veryimportant{零级}\\
    \verbroot{rudh} → 零级 \verbstem{rudh-} + s → \fullpada{aruddha}
    \item s 在词内连声中变化明显，需注意连声：\\
    \fullpada{arautsam} ~~\fullpada{arautsīt} ~~\fullpada{arautsuḥ}
    \item \verbroot{nī} 'lead' → \fullpada{anaiṣam} ~~\fullpada{anaiṣīt} ~~\fullpada{anaiṣuḥ}
    \item \verbroot{bhū} 'be' → \fullpada{abhūṣam} ~~\fullpada{abhūṣīt}
  \end{itemize}
\end{frame}

\begin{frame}{重复不定过去时（2）~~致使动词}
  \small
  \begin{itemize}
    \item 主要用于\important{派生动词}（致使动词、第十类动词）
    \item 重复规则（类似完成时，→ 第27--28章）：\\
    i/ī → 重复 \pratyaya{i}，u/ū → 重复 \pratyaya{u}\\
    a/ṛ → 轻音节重复 \pratyaya{a}，重音节重复 \pratyaya{ā}
    \item 语干：增音符 + 重复音节 + 词根（次浊/零级）+ \pratyaya{-a-}\\
    \verbroot{jan}（致使 jānaya-）：\\
    ~~\pratyaya{a-} + \pratyaya{jī-} + \verbstem{jan-} + a → \fullpada{ajījanam}\\
    \verbroot{kṛ}（致使 kārayati）：\\
    ~~\pratyaya{a-} + \pratyaya{cī-} + \verbstem{kār-} + a → \fullpada{acīkarat}
    \item \important{区别于完成时}：重复不定过去时有增音符，\\
    完成时无增音符（\fullpada{cakāra} vs. \fullpada{acīkarat}）
  \end{itemize}
\end{frame}

\begin{frame}{\skt{mā} + 不定过去时~~禁止命令}
  \small
  \begin{itemize}
    \item \skt{mā} + \veryimportant{不含增音符}的不定过去时\\
    = 禁止某事（否定命令）
    \item 这是不定过去时在古典梵语中\important{最常见的用法}。
    \bigskip
    \item \skt{mā bhūḥ} ~~（√bhū，词根不定过去时 2nd.Sg.）\\
    \notsoimportant{"别（这样）存在/做！"}
    \item \skt{mā śucaḥ} ~~（√śuc，a-不定过去时 2nd.Sg.）\\
    \notsoimportant{"别悲哀！"}~~（参见《薄伽梵歌》第38章阅读）
    \item \skt{mā bhūt} ~~（3rd.Sg.）~~\notsoimportant{"别让……！"}
    \bigskip
    \item 对比第24章：\skt{na} + 命令式（更强硬的禁止）\\
    \notsoimportant{两种否定命令在古典文学中均可见}
  \end{itemize}
\end{frame}

\begin{frame}{\insertsection}
  \footnotesize
  \centering
  \begin{NiceTabular}{|c|c|c|c|}[hvlines, rules/width=0.3pt, rules/color=gray]
    & \important{语干} & \important{语尾} & \important{增音符} \\
    不完全过去时 & 现在时语干 & 次要语尾 & 有 \\
    完成时（第27--28章） & 重复+完成时语干 & 完成时语尾 & 无 \\
    迂回完成时（第34章） & 现在时语干+\pratyaya{-ām}+助动词 & — & 无 \\
    \important{不定过去时} & \important{不定过去时语干（多种）} & \important{次要语尾} & \important{有} \\
  \end{NiceTabular}
  \vspace{0.5em}
  \begin{itemize}
    \item 有增音符 + 次要语尾 + 非现在时语干 → \important{不定过去时}
    \item s/iṣ 等后缀出现 → \important{s 类不定过去时}
    \item 重复音节 + 增音符同时出现 → \important{重复不定过去时}（派生动词）
  \end{itemize}
\end{frame}
