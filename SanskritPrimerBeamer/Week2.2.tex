\section{名词构成}

\begin{frame}{名词的构成}
  \small
  \begin{itemize}
    \item 复合词：\\两个或多个名词组合构成（14章）
    \item 其他词+词缀构成
    \begin{itemize}
      \item 动词词根+原始词缀（kṛt）
      \item 名词性词+派生词缀（taddhita）
    \end{itemize}
  \end{itemize}
\end{frame}

\subsection{原始词缀}
\begin{frame}{\insertsubsection ~~\pratyaya{-a-}}
  \small
  \begin{itemize}
    \item 动词词根元音二合或a三合
    \item 构成阳性名词表示行为或状态
    \item 构成形容词或名词表示施事者
    \bigskip
    \\ \verbroot{likh} → \nounstem{lekha} m. 书写、信件
    \\ \verbroot{vid} → \nounstem{veda} m. 知识、吠陀
    \\ \verbroot{hṛṣ} → \nounstem{harṣa} m. 兴奋；adj. 令人兴奋的
    \\ \verbroot{hṛ} → \nounstem{hara}/\nounstem{hāra} m. 携带、负担
    \\ \verbroot{budh} → \nounstem{bodha} m. 觉醒、意识；\\~~~~~~~~~~~~~~~~~~~~~~adj. 知晓的
  \end{itemize}
\end{frame}

\begin{frame}{\insertsubsection~~\pratyaya{-ana-}}
  \small
  \begin{itemize}
    \item 动词词根元音二合
    \item 构成中性名词表达行为或状态（\textit{-ing}）
    \item 或者行为或状态所凭借发生的途径或方式
    \item \nobreakdash-ana- 形容词常作复合词后词
    \bigskip
    \\ \verbroot{vac} → \nounstem{vacana} n. 言语、词语
    \\ \verbroot{muc} → \nounstem{mocana} n. 释放；adj. 解放的
    \\ \verbroot{śubh} → \nounstem{śobhana} n. 美丽，吉祥
    \\ \verbroot{mṛ} → \nounstem{maraṇa} n. 死亡
    \\ \verbroot{loc} → \nounstem{locana} n. 眼睛（看之工具）
  \end{itemize}
\end{frame}

%==================================================

\begin{frame}{\insertsubsection~~\pratyaya{-tra-}}
  \small
  \begin{itemize}
    \item 动词词根元音二合
    \item 构成中性名词表示行为发生的工具或方式
    \bigskip
    \\ \verbroot{śās} → \nounstem{śāstra} n. 教令、典籍、教科书
    \\ \verbroot{śas} → \nounstem{śastra} n. 刀、剑（切割工具）
    \\ \verbroot{man} → \nounstem{mantra} n. 咒语、祈文
    \\ \verbroot{nī} → \nounstem{netra} n. 眼睛（引导工具）
    \\ \verbroot{pat} → \nounstem{patra} n. 羽翼（飞翔工具）
  \end{itemize}
\end{frame}

\subsection{派生词缀}

\begin{frame}{\insertsubsection~~\pratyaya{-tva-}}
  \small
  \begin{itemize}
    \item 构成中性抽象名词表示某种状态或性质
    \bigskip
    \\ \nounstem{nara} → \nounstem{naratva} n. 人性、做人
    \\ \nounstem{deva} → \nounstem{devatva} n. 神性、成为神
    \\ \nounstem{guru} → \nounstem{gurutva} n. 为师之道
    \\ \nounstem{abuddha} → \nounstem{abuddhatva} n. 愚蠢
  \end{itemize}
\end{frame}

\begin{frame}{\insertsubsection~~\pratyaya{-tā-}}
  \small
  \begin{itemize}
    \item 构成阴性抽象名词表示某种状态或性质
    \bigskip
    \\ \nounstem{priya} → \nounstem{priyatā} f. 可爱、亲爱之情
    \\ \nounstem{martya} → \nounstem{martyatā} f. 终有一死
    \\ \nounstem{mitra} → \nounstem{mitratā} f. 友谊
    \\ \nounstem{puruṣa} → \nounstem{puruṣatā} f. 阳刚、男性气质
    \\ \nounstem{deva} → \nounstem{devatā} f. 神性、天神
    \bigskip
    \item \notsoimportant{部分词同时有 \pratyaya{-tva-} 和 \pratyaya{-tā-} 两种形式}
  \end{itemize}
\end{frame}

%===================================================

\begin{frame}{\insertsubsection~~\pratyaya{-a-/-ya-}}
  \small
  \begin{itemize}
    \item 加 \pratyaya{-a-} 时名词的第一个元音通常三合
    \item 加 \pratyaya{-ya-} 时可能去掉末尾元音，较少三合
    \item 表示"属于……的"，"与……有关的"
    \bigskip
    \\ \nounstem{pura} 城市 → \nounstem{paura} m. 市民
    \\ \nounstem{veda} 知识 → \nounstem{vaidya} m. 医生
    \\ \nounstem{senā} 军队 → \nounstem{sainya} m. 士兵
    \\ \nounstem{śūra} 英雄 → \nounstem{śaurya} n. 英勇、武勇
    \\ \nounstem{mitra} 朋友 → \nounstem{maitra}/\nounstem{maitrī} f. 友谊
  \end{itemize}
\end{frame}

\begin{frame}{\insertsubsection~~\pratyaya{-a-}}
  \begin{itemize}
    \item 由地名派生表示该地居民  
    \\ \nounstem{niṣadha} → \nounstem{naiṣadha} 尼沙陀国王
    \\ \nounstem{vidharbha} → \nounstem{vaidarbhī} 不毛国公主
    \item 由人名派生表示后代、子孙
    \\ \nounstem{kuru} → \fullpada{kauravāḥ} 俱卢族 
    \\ \nounstem{pāṇḍu} → \fullpada{pāṇḍavāḥ} 般度族
    \\ \nounstem{raghu} → \nounstem{rāghava} 罗祜后裔
  \end{itemize}
\end{frame}

\begin{frame}{总结}
  \centering
  \small
  \begin{NiceTabular}{lll}
    \textbf{词缀} & \textbf{主要变化} & \textbf{结果} \\
    原始词缀 & & \\
    \pratyaya{-a-} & 词根二合 & 阳性名词 / 形容词 \\
    \pratyaya{-ana} & 词根二合 & 中性名词 / 形容词 \\
    \pratyaya{-tra} & 词根二合 & 中性工具名词 \\
    派生词缀 & & \\
    \pratyaya{-tva} & 不变 & 中性抽象名词 \\
    \pratyaya{-tā} & 不变 & 阴性抽象名词 \\
    \pratyaya{-a-/-ya-} & 首元音三合 & "属于……的" \\
  \end{NiceTabular}
  \begin{itemize}
    \item 便于减轻记忆压力，拿不准还是查字典 
  \end{itemize}
\end{frame}
