    \item 构成\textbf{阳性名词}和形容词（偶尔中性）
    \bigskip
    \item \verbroot{likh} → \nounstem{lekha} (m.)~~书写、信件
    \item \verbroot{vid} → \nounstem{veda} (m.)~~知识、吠陀
    \item \verbroot{hṛṣ} → \nounstem{harṣa} (m.)~~兴奋；(Adj) 令人兴奋的
    \item \verbroot{bhṛ} → \nounstem{bhara}/\nounstem{bhāra} (m.)~~携带、负担
    \item \verbroot{budh} → \nounstem{bodha} (m.)~~觉醒、意识；(Adj) 知晓的
  \end{itemize}
\end{frame}

\subsection{ana}

\begin{frame}{\insertsubsection}

\subsection{tra}

\begin{frame}{\insertsubsection}

%===================================================
\subsection{tva与tā}

\begin{frame}{tva~~中性抽象名词}
  \small
  \begin{itemize}
    \item 加在\textbf{名词或形容词}上（不加在词根上）
    \item 意义："……的状态、性质"（≈ 英语 \textit{-ness, -hood}）
    \item 构成\textbf{中性}名词
    \bigskip
    \item \nounstem{nara} → \nounstem{naratva} (n.)~~人性、做人
    \item \nounstem{deva} → \nounstem{devatva} (n.)~~神性、成为神
    \item \nounstem{guru} → \nounstem{gurutva} (n.)~~为师之道
    \item \nounstem{abuddha} → \nounstem{abuddhatva} (n.)~~愚蠢
  \end{itemize}
\end{frame}

\begin{frame}{tā~~阴性抽象名词}
  \small
  \begin{itemize}
    \item 同样加在\textbf{名词或形容词}上，意义与 \wordending{tva} 相近
    \item 构成\textbf{阴性}名词
    \bigskip
    \item \nounstem{priya} → \nounstem{priyatā} (f.)~~可爱、亲爱之情
    \item \nounstem{mitra} → \nounstem{mitratā} (f.)~~友谊
    \item \nounstem{puruṣa} → \nounstem{puruṣatā} (f.)~~阳刚、男性气质
    \item \nounstem{deva} → \nounstem{devatā} (f.)~~神性；神灵（可指具体神）
    \bigskip
    \item \notsoimportant{部分词同时有 \wordending{tva} 和 \wordending{tā} 两种形式}
  \end{itemize}
\end{frame}

\section{三合派生词}

\begin{frame}{\insertsection~~构成}
  \small
  \begin{itemize}
    \item 将词的\textbf{第一个元音}变为 vṛddhi 级
    \item 若非 \nobreakdash-a\nobreakdash- 词干，加 \wordending{ya}；末尾 \nobreakdash-a\nobreakdash- 也可换为 \wordending{ya}
    \item 意义：\important{"属于……的"，"与……有关的"}
    \bigskip
    \item \nounstem{pura}（城市）→ \nounstem{paura} (m.)~~市民
    \item \nounstem{veda}（知识）→ \nounstem{vaidya} (m.)~~医生
    \item \nounstem{senā}（军队）→ \nounstem{sainya} (m.)~~士兵
    \item \nounstem{śūra}（英雄）→ \nounstem{śaurya} (n.)~~英勇、武勇
    \item \nounstem{mitra}（朋友）→ \nounstem{maitra}/\nounstem{maitrī} (f.)~~友谊
  \end{itemize}
\end{frame}

\begin{frame}{Vṛddhi 派生词——专有名词}
  \begin{itemize}
    \item \textbf{地名} → 该地居民
    \item \textbf{人名} → 后代、子孙
    \bigskip
    \item Niṣadha（地名）→ \nounstem{naiṣadha}~~"来自尼沙达的人"
    \item Kuru（人名）→ \nounstem{kaurava}~~"俱卢族"
    \item Pāṇḍu（人名）→ \nounstem{pāṇḍava}~~"般度族"
    \item Raghu（人名）→ \nounstem{rāghava}~~"拉格胡之后"（即罗摩）
    \bigskip
    \item \notsoimportant{《摩诃婆罗多》中 Kaurava 对阵 Pāṇḍava}
  \end{itemize}
\end{frame}
