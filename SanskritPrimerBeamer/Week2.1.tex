\begin{frame}{复合词}
  \small
  传统上，梵语复合词分为三大类：
  \begin{itemize}
    \item 并列复合词 (相违释 / Dvandva)
    \item 限定复合词 (依主释 / Tatpuruṣa)
    \begin{itemize}
      \item 同位复合词\\ (持业释 / Karmadhāraya)
    \end{itemize}
    \item 定语复合词 (多财释 / Bahuvrīhi)
  \end{itemize}
\end{frame}

\subsection{词内连声}

\begin{frame}{\insertsubsection ~~buddha连声}
  \begin{itemize}
    \item 送气浊音之后，t和th变成dh
    \item 浊辅音之前的辅音变成浊不送气
    \begin{itemize}
      \item \verbroot{budh} + \pratyaya{ta} = \nounstem{buddha}
      \item \verbroot{duh} + \pratyaya{ta} = \nounstem{dugdha}
      \item 反例：\verbroot{dhā} + \wordending{thas}(p2d) =  \fullpada{dhatthaḥ}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{\insertsubsection ~~ruki规则}
  \begin{itemize}
    \item 前音是非a/ā元音、半元音、h或者喉音时，s变成ṣ
    \item 前音和s中间插入了ṃ、ḥ或者咝音仍要变化
    \begin{itemize}
      \item \nounstem{agni} + \wordending{su} = \fullpada{agniṣu}
      \item \nounstem{yajuḥ} + \wordending{su} = \fullpada{yajuḥṣu}
      \item 反例：词尾不变 \fullsentence{agnis tatra}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{\insertsubsection ~~n的顶化}
  \begin{itemize}
    \item 词内前方有ṛ、r或ṣ时，n变成ṇ
    \item 中间有元音、y、v、ṃ、ḥ、喉音和唇音仍要变化
    \begin{itemize}
      \item \nounstem{mitra}(G.pl.) = \fullpada{mitrāṇām}
      \item 反例：词尾不变 \fullpada{brahman}, \fullpada{brahmaṇā}
      \item 反例：\nounstem{ratha}(I.sg.) = \fullpada{rathena}
    \end{itemize}
  \end{itemize}
\end{frame}
